function usersKey(){ return 'jee_wr_users_v1'; }
function dataKey(u){ return 'jee_wr_data_' + u.toLowerCase(); }

function loadUsers(){
  try { return JSON.parse(localStorage.getItem(usersKey())||'{}'); } catch(e){ return {}; }
}
function saveUsers(u){ localStorage.setItem(usersKey(), JSON.stringify(u)); }

function defaultStore(){
  return {
    tasks:[], flashcards:[], studyLog:[], revision:[], notes:[], formulas:[],
    pyq:[], mocks:[], mistakes:[], weakTopics:[],
    subjects:{phy:0,chem:0,math:0},
    syllabus:{phy:{},chem:{},math:{}},
    calendar:{}, streak:0, lastStudyDate:null, todayMinutes:0
  };
}

function loadStore(u){
  try {
    const raw = localStorage.getItem(dataKey(u));
    if(!raw) return defaultStore();
    return Object.assign(defaultStore(), JSON.parse(raw));
  } catch(e){ return defaultStore(); }
}
function saveStore(){
  if(!currentUser) return;
  localStorage.setItem(dataKey(currentUser), JSON.stringify(store));
}

function showAuthError(msg){
  const el = $('authError');
  el.textContent = msg;
  el.classList.add('show');
}
function clearAuthError(){ $('authError').classList.remove('show'); $('authError').textContent=''; }

function login(username, password){
  const users = loadUsers();
  const key = username.toLowerCase().trim();
  if(!users[key]) return showAuthError('User nahi mila.');
  if(users[key].password !== password) return showAuthError('Password galat hai.');
  currentUser = key;
  store = loadStore(key);
  enterApp();
}
function signup(username, password){
  const users = loadUsers();
  const key = username.toLowerCase().trim();
  if(key.length < 2) return showAuthError('Username chhota hai.');
  if(password.length < 4) return showAuthError('Password min 4 char.');
  if(users[key]) return showAuthError('Username already exists.');
  users[key] = { password, createdAt: Date.now() };
  saveUsers(users);
  currentUser = key;
  store = defaultStore();
  saveStore();
  enterApp();
}

function enterApp(){
  $('authModal').classList.add('hide');
  $('app').style.display = '';
  $('userNameDisplay').textContent = currentUser;
  $('userAvatar').textContent = (currentUser[0]||'?').toUpperCase();
  updateStreak();
  renderAll();
  updateCountdown();
  setInterval(updateCountdown, 60000);
  const q = QUOTES[Math.floor(Math.random()*QUOTES.length)];
  $('motivationQuote').textContent = '"'+q.q+'"';
  $('motivationAuthor').textContent = '— '+q.a;
}

function logout(){
  currentUser = null; store = null;
  $('app').style.display = 'none';
  $('authModal').classList.remove('hide');

}

function updateCountdown(){
  // Next JEE Main approx Jan 2027 session
  const target = new Date('2027-01-28T09:00:00+05:30');
  const now = new Date();
  const diff = target - now;
  if(diff <= 0){ $('jeeCountdown').textContent='LIVE / PASSED'; $('daysLeft').textContent='0'; return; }
  const days = Math.floor(diff/86400000);
  const hrs = Math.floor((diff%86400000)/3600000);
  const mins = Math.floor((diff%3600000)/60000);
  $('jeeCountdown').textContent = days+'d '+hrs+'h '+mins+'m';
  $('daysLeft').textContent = String(days);
  $('jeeDateSub').textContent = 'Target: 28 Jan 2027 (approx)';
}

function updateStreak(){
  const today = new Date().toISOString().slice(0,10);
  if(store.lastStudyDate === today){
    // keep
  } else {
    const y = new Date(); y.setDate(y.getDate()-1);
    const ys = y.toISOString().slice(0,10);
    if(store.lastStudyDate === ys){
      // continued
    } else if(store.lastStudyDate && store.lastStudyDate !== today){
      // broken handled when studying
    }
  }
  $('streakNum').textContent = store.streak || 0;
}

function markStudy(minutes){
  const today = new Date().toISOString().slice(0,10);
  store.todayMinutes = (store.calendar[today]||0);
  if(store.lastStudyDate !== today){
    const y = new Date(); y.setDate(y.getDate()-1);
    const ys = y.toISOString().slice(0,10);
    if(store.lastStudyDate === ys) store.streak = (store.streak||0)+1;
    else store.streak = 1;
    store.lastStudyDate = today;
  }
  store.calendar[today] = (store.calendar[today]||0) + minutes;
  store.todayMinutes = store.calendar[today];
  saveStore();
  updateStreak();
}

function renderDashboard(){
  const pending = (store.tasks||[]).filter(t=>!t.done).length;
  $('statTasks').textContent = pending;
  const today = new Date().toISOString().slice(0,10);
  const tm = store.calendar[today]||0;
  $('statTime').textContent = tm >= 60 ? (tm/60).toFixed(1)+'h' : tm+'m';
  const totalH = (store.subjects.phy+store.subjects.chem+store.subjects.math);
  $('statTotal').textContent = totalH.toFixed(1)+'h';
  $('statMocks').textContent = (store.mocks||[]).length;
  const pct = Math.min(100, (tm/480)*100);
  $('dailyProgress').style.width = pct+'%';
  $('dailyProgressText').textContent = (tm/60).toFixed(1)+' / 8 hours';
  $('statFlash').textContent = (store.flashcards||[]).length;
  const now = Date.now();
  $('statRevDue').textContent = (store.revision||[]).filter(r=>r.nextDate<=now).length;
  $('statWeak').textContent = (store.weakTopics||[]).length;
  $('streakNum').textContent = store.streak||0;
}

function renderSyllabus(){
  ['phy','chem','math'].forEach(sub=>{
    const box = $('syl'+sub[0].toUpperCase()+sub.slice(1));
    const chs = SYLLABUS[sub];
    const done = chs.filter(c=>store.syllabus[sub][c]).length;
    const pct = Math.round(done/chs.length*100);
    $('syl'+sub[0].toUpperCase()+sub.slice(1)+'Pct').textContent = pct+'%';
    box.innerHTML = chs.map(c=>`
      <label class="chapter ${store.syllabus[sub][c]?'done':''}">
        <input type="checkbox" ${store.syllabus[sub][c]?'checked':''} data-sub="${sub}" data-ch="${esc(c)}">
        <span>${esc(c)}</span>
      </label>`).join('');
  });
  const all = [...SYLLABUS.phy,...SYLLABUS.chem,...SYLLABUS.math];
  const allDone = all.filter(c=>store.syllabus.phy[c]||store.syllabus.chem[c]||store.syllabus.math[c]).length;
  // fix overall
  let d=0,t=0;
  ['phy','chem','math'].forEach(s=>{t+=SYLLABUS[s].length;d+=SYLLABUS[s].filter(c=>store.syllabus[s][c]).length;});
  $('sylTotalPct').textContent = Math.round(d/t*100)+'%';
  document.querySelectorAll('#sylPhy input, #sylChem input, #sylMath input').forEach(el=>{
    el.onchange=()=>{store.syllabus[el.dataset.sub][el.dataset.ch]=el.checked;saveStore();renderSyllabus();};
  });
}

function renderCalendar(){
  const y=calDate.getFullYear(), m=calDate.getMonth();
  $('calMonth').textContent = calDate.toLocaleString('en',{month:'long',year:'numeric'});
  const first=new Date(y,m,1).getDay();
  const days=new Date(y,m+1,0).getDate();
  const today=new Date().toISOString().slice(0,10);
  let html='';
  for(let i=0;i<first;i++) html+='<div class="cal-day"></div>';
  for(let d=1;d<=days;d++){
    const key=`${y}-${String(m+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
    const mins=store.calendar[key]||0;
    const cls=['cal-day'];
    if(mins>0) cls.push('studied');
    if(key===today) cls.push('today');
    html+=`<div class="${cls.join(' ')}"><div>${d}</div>${mins?`<div style="font-size:.55rem">${mins}m</div>`:''}</div>`;
  }
  $('calGrid').innerHTML=html;
}

function fmtTime(s){ const m=Math.floor(s/60), ss=s%60; return String(m).padStart(2,'0')+':'+String(ss).padStart(2,'0'); }
function tickTimer(){
  if(timerLeft<=0){
    clearInterval(timerRun); timerRun=null;
    if(timerMode==='Focus'){ markStudy(timerSec/60); const sub=prompt('Kaunsa subject? phy/chem/math','phy'); if(sub&&store.subjects[sub]!=null){store.subjects[sub]+=(timerSec/60);saveStore();renderSubjects();renderDashboard();} }
    alert(timerMode+' complete!');
    return;
  }
  timerLeft--; $('timerDisplay').textContent=fmtTime(timerLeft);
}
